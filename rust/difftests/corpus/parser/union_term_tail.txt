## A bare (unparenthesized) NON-FINAL term of a UNION/EXCEPT/INTERSECT
## chain can carry its own trailing ORDER BY/LIMIT, disambiguated by the
## set operator immediately following (confirmed via godump restore) --
## this used to be a genuine parser bug here ("unexpected trailing
## tokens"), fixed alongside FOR UPDATE's own implementation since both
## needed the same tail-parsing restructuring.
select a from t order by a union select b from t2
select a from t limit 1 union select b from t2
select a from t order by a limit 1 union select b from t2
select a from t union select b from t2 order by a union select c from t3
## Whether a term is parenthesized or bare makes no difference to where
## a trailing statement-level tail after the LAST term attaches.
select a from t1 union all (select a from t2) order by 1 limit 10
(select a from t1) union all select a from t2 order by 1 limit 10
(select a from t1) union all (select a from t2) order by 1 limit 10
