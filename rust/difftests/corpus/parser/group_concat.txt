## GROUP_CONCAT: SEPARATOR always restores explicitly (default ',') even
## when not written; multiple args concatenate per row like CONCAT
select group_concat(a) from t group by b
select group_concat(distinct a) from t group by b
select group_concat(a separator '-') from t group by b
select group_concat(a, b) from t group by c
select group_concat(a separator ', ') from t
