## TiDB registers `sql_safe_updates` in `variable/noop.go`: it is typed
## session/global compatibility state, not a MySQL-style UPDATE/DELETE guard.
## `SET_VAR(sql_safe_updates=...)` is the one source-backed hint overlay:
## scoped to its SELECT and restored even when SELECT evaluation fails. Other
## parsed hint kinds keep the executor's current no-effect behavior.
select @@sql_safe_updates

set sql_safe_updates=1
select @@sql_safe_updates
set sql_safe_updates=3
select @@sql_safe_updates
set sql_safe_updates=default
select @@sql_safe_updates
select /*+ SET_VAR(sql_safe_updates=1) */ @@sql_safe_updates
select @@sql_safe_updates
select /*+ SET_VAR(sql_safe_updates=1) SET_VAR(sql_safe_updates=0) */ @@sql_safe_updates
select @@sql_safe_updates
