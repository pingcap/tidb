
#[test]
fn audit_workspace_structural_boundaries() {
    let (mut session, _) = open_session();
    for sql in [
        "SELECT id FROM t",
        "SELECT 1 UNION ALL SELECT 2",
        "SELECT missing_column FROM t",
        "SELECT * FROM missing_table",
    ] {
        match session.prepare_general(sql) {
            Ok(prepared) => println!("AUDIT prepare\t{sql}\tok\t{}", prepared.result_columns().len()),
            Err(error) => println!("AUDIT prepare\t{sql}\terror\t{error:?}"),
        }
    }
    let mut local = tidb_session::Session::new();
    for sql in [
        "CREATE SEQUENCE audit_sequence START WITH 7",
        "SELECT NEXTVAL(audit_sequence)",
        "SELECT COUNT(*) FROM information_schema.sequences WHERE sequence_name = 'audit_sequence'",
        "SELECT * FROM information_schema.cluster_config LIMIT 1",
        "SET tidb_snapshot = '2026-09-01 00:00:00'",
        "SELECT 1",
    ] {
        println!("AUDIT session\t{sql}\t{:?}", local.run(sql));
    }
}
