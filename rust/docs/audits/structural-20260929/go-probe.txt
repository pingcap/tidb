package executor_test

import (
    "testing"
    "github.com/pingcap/tidb/pkg/testkit"
)

func TestRustWorkspaceStructuralBoundaries(t *testing.T) {
    store := testkit.CreateMockStore(t)
    tk := testkit.NewTestKit(t, store)
    tk.MustExec("use test")
    tk.MustExec("create table t(id bigint primary key, v bigint)")
    for _, sql := range []string{
        "SELECT id FROM t",
        "SELECT 1 UNION ALL SELECT 2",
        "SELECT missing_column FROM t",
        "SELECT * FROM missing_table",
    } {
        id, _, fields, err := tk.Session().PrepareStmt(sql)
        if err != nil { t.Logf("AUDIT prepare\t%s\terror\t%v", sql, err) } else {
            t.Logf("AUDIT prepare\t%s\tok\t%d", sql, len(fields))
            if err := tk.Session().DropPreparedStmt(id); err != nil { t.Fatal(err) }
        }
    }
    tk.MustExec("CREATE SEQUENCE audit_sequence START WITH 7")
    tk.MustQuery("SELECT NEXTVAL(audit_sequence)").Check(testkit.Rows("7"))
    tk.MustQuery("SELECT COUNT(*) FROM information_schema.sequences WHERE sequence_name = 'audit_sequence'").Check(testkit.Rows("1"))
    t.Log("AUDIT sequence\tnextval=7\tmetadata_count=1")
}
