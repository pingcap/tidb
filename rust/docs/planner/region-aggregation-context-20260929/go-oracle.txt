 ft:=func(domain string)*tipb.FieldType{
  if domain=="UInt" {domain="Int"}
  code:=map[string]byte{"Int":mysql.TypeLonglong,"Real":mysql.TypeDouble,"Float":mysql.TypeFloat,"Decimal":mysql.TypeNewDecimal,"String":mysql.TypeVarString,"Time":mysql.TypeDatetime,"Duration":mysql.TypeDuration,"Json":mysql.TypeJSON}[domain]
  field:=&tipb.FieldType{Tp:int32(code),Flen:-1,Decimal:-1,Charset:"binary",Collate:63,Flag:128}
  if domain=="String"||domain=="Json" {field.Charset="utf8mb4";field.Collate=46;field.Flag=0}
  if domain=="Int" {field.Decimal=0};if domain=="Decimal" {field.Flen=10;field.Decimal=2};if domain=="Time"||domain=="Duration" {field.Decimal=3}
  return field
 }
 var literal func(string,string)*tipb.Expr
 literal=func(domain,value string)*tipb.Expr {
  pb:=&tipb.Expr{FieldType:ft(domain)}
  switch domain {
  case "Null": pb.Tp=tipb.ExprType_Null
  case "Int":n,_:=strconv.ParseInt(value,10,64);pb.Tp=tipb.ExprType_Int64;pb.Val=codec.EncodeInt(nil,n)
  case "UInt":n,_:=strconv.ParseUint(value,10,64);pb.FieldType=ft("Int");pb.FieldType.Flag|=32;pb.Tp=tipb.ExprType_Uint64;pb.Val=codec.EncodeUint(nil,n)
  case "Float":n,_:=strconv.ParseFloat(value,32);pb.Tp=tipb.ExprType_Float32;pb.Val=codec.EncodeFloat(nil,float64(float32(n)))
  case "Real":n,_:=strconv.ParseFloat(value,64);pb.Tp=tipb.ExprType_Float64;pb.Val=codec.EncodeFloat(nil,n)
  case "Decimal":d:=new(types.MyDecimal);if err:=d.FromString([]byte(value));err!=nil {panic(err)};pb.Tp=tipb.ExprType_MysqlDecimal;pb.Val,err=codec.EncodeDecimal(nil,d,0,0)
  case "String":pb.Tp=tipb.ExprType_String;pb.Val=[]byte(value)
  case "Duration":n,_:=strconv.ParseInt(value,10,64);pb.Tp=tipb.ExprType_MysqlDuration;pb.Val=codec.EncodeInt(nil,n)
  case "Time":pb.Tp=tipb.ExprType_MysqlTime;day,_:=strconv.Atoi(value);v:=types.NewTime(types.FromDate(2024,1,day,3,4,5,123000),mysql.TypeDatetime,3);n,_:=v.ToPackedUint();pb.Val=codec.EncodeUint(nil,n)
  case "Json":pb.Tp=tipb.ExprType_MysqlJson;j,e:=types.ParseBinaryJSONFromString(value);if e!=nil {panic(e)};pb.Val,err=codec.EncodeValue(time.UTC,nil,types.NewJSONDatum(j))
  }
  return pb
 }
