out=[]

Path('/private/tmp/positional-differences.json').write_text(json.dumps(out,indent=2)+'\n')

 if all(not a.startswith('s') or (lambda b: b.decode('utf8','ignore').encode()==b)(bytes.fromhex(a[1:])) for a in x['args']):valid.append(x)
