 out,e:=os.Create("/private/tmp/batch-cast-go.tsv");if e!=nil{t.Fatal(e)};defer out.Close()
 for _,su:=range []bool{false,true}{for _,tu:=range []bool{false,true}{for _,flen:=range []int{10,30}{for _,value:=range []int64{-1,0,1,math.MinInt64,math.MaxInt64}{for _,flags:=range []uint64{0,1,2}{for _,route:=range []string{"scalar","batch","suite_on","suite_off"}{func(){
  ctx:=mock.NewContextDeprecated();ctx.GetSessionVars().StmtCtx.InitFromPBFlagAndTz(flags,time.UTC);ctx.GetSessionVars().TimeZone=time.UTC

 }()}}}}}}
