func TestTransportContractReadiness(t *testing.T) {
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	require.NoError(t, err)
	defer listener.Close()
	setup, cancel := context.WithCancel(context.Background())
	conn, err := GetClientConn(setup, "http://"+listener.Addr().String(), nil)
	require.NoError(t, err)
	defer conn.Close()
	cancel()
	require.NoError(t, listener.(*net.TCPListener).SetDeadline(time.Now().Add(3*time.Second)))
	socket, err := listener.Accept()
	require.NoError(t, err, "nonblocking dial must connect without an RPC and outlive setup cancellation")
	defer socket.Close()
	preface := make([]byte, 24)
	require.NoError(t, socket.SetReadDeadline(time.Now().Add(3*time.Second)))
	_, err = io.ReadFull(socket, preface)
	require.NoError(t, err)
	require.Equal(t, "PRI * HTTP/2.0\r\n\r\nSM\r\n\r\n", string(preface))

	// A second peer accepts TCP and reads the client preface, but withholds SETTINGS.
	accepted := make(chan struct{})
	stop := make(chan struct{})
	done := make(chan struct{})
	go func() {
		defer close(done)
		peer, acceptErr := listener.Accept()
		if acceptErr != nil {
			return
		}
		defer peer.Close()
		if _, readErr := io.ReadFull(peer, make([]byte, 24)); readErr != nil {
			return
		}
		close(accepted)
		<-stop
	}()
	defer func() { close(stop); listener.Close(); <-done }()
	ctx, cancelBlocking := context.WithTimeout(context.Background(), 500*time.Millisecond)
	defer cancelBlocking()
	blocked, err := GetClientConn(ctx, "http://"+listener.Addr().String(), nil, grpc.WithBlock())
	require.Error(t, err)
	require.Nil(t, blocked)
	require.Contains(t, err.Error(), "PD:grpc:ErrGRPCDial")
	select {
	case <-accepted:
	default:
		t.Fatal("blocking dial never reached the stalled HTTP/2 peer")
	}
	require.NotEqual(t, connectivity.Ready, conn.GetState())
	output := os.Getenv("PD_GRPCUTIL_ORACLE")
	require.NotEmpty(t, output)
	data, err := json.MarshalIndent(map[string]bool{"lazy_connects_in_background": true, "connect_waits_for_server_settings": true}, "", "  ")
	require.NoError(t, err)
	require.NoError(t, os.WriteFile(filepath.Join(output, "go-readiness.json"), append(data, '\n'), 0600))
}

func TestTransportContractMetadata(t *testing.T) {
	base := metadata.NewOutgoingContext(context.Background(), metadata.Pairs("existing", "value"))
	forward := BuildForwardContext(base, "https://pd:2379")
	require.Equal(t, "https://pd:2379", GetForwardedHost(forward, metadata.FromOutgoingContext))
	require.Empty(t, GetForwardedHost(forward, metadata.FromIncomingContext))
	md, _ := metadata.FromOutgoingContext(forward)
	require.Equal(t, metadata.Pairs(ForwardMetadataKey, "https://pd:2379"), md)
	follower := BuildFollowerHandleContext(forward)
	require.True(t, IsFollowerHandleEnabled(follower, metadata.FromOutgoingContext))
	require.Empty(t, GetForwardedHost(follower, metadata.FromOutgoingContext))
	require.False(t, IsFollowerHandleEnabled(context.Background(), metadata.FromOutgoingContext))
	incoming := metadata.NewIncomingContext(context.Background(), metadata.Pairs(ForwardMetadataKey, "first", ForwardMetadataKey, "second"))
	require.Equal(t, "first", GetForwardedHost(incoming, metadata.FromIncomingContext))
	require.Panics(t, func() {
		GetForwardedHost(context.Background(), func(context.Context) (metadata.MD, bool) { return metadata.MD{ForwardMetadataKey: []string{}}, true })
	})
}

func TestTransportContractRetryOwnership(t *testing.T) {
	terminal := status.Error(codes.Unavailable, "last RPC error")
	ctx, cancel := context.WithCancel(context.Background())
	ctx = retry.WithBackoffer(ctx, retry.InitialBackoffer(time.Hour, time.Hour, 0))
	calls := 0
	invoke := func(context.Context, string, any, any, *grpc.ClientConn, ...grpc.CallOption) error {
		calls++
		cancel()
		return terminal
	}
	err := UnaryBackofferInterceptor()(ctx, "/test", nil, nil, nil, invoke)
	require.Same(t, terminal, err, "cancellation while backing off returns the last RPC error")
	require.Equal(t, 1, calls)
	shared := retry.InitialBackoffer(time.Millisecond, time.Millisecond, 2*time.Millisecond)
	sharedCtx := retry.WithBackoffer(context.Background(), shared)
	var wg sync.WaitGroup
	for range 8 {
		wg.Add(1)
		go func() {
			defer wg.Done()
			attempts := 0
			err := UnaryBackofferInterceptor()(sharedCtx, "/test", nil, nil, nil, func(context.Context, string, any, any, *grpc.ClientConn, ...grpc.CallOption) error {
				attempts++
				if attempts == 1 {
					return terminal
				}
				return nil
			})
			require.NoError(t, err)
			require.Equal(t, 2, attempts)
		}()
	}
	wg.Wait()
	shared.SetRetryableChecker(func(error) bool { return false }, true)
	calls = 0
	err = UnaryBackofferInterceptor()(sharedCtx, "/test", nil, nil, nil, func(context.Context, string, any, any, *grpc.ClientConn, ...grpc.CallOption) error {
		calls++
		return terminal
	})
	require.Same(t, terminal, err)
	require.Equal(t, 1, calls)
}

func TestTransportContractBreaker(t *testing.T) {
	for code := codes.OK; code <= codes.Unauthenticated; code++ {
		overloaded := code == codes.DeadlineExceeded || code == codes.Unavailable || code == codes.ResourceExhausted
		err := status.Error(code, "test")
		require.Equal(t, overloaded, bool(isOverloaded(err)))
		require.Equal(t, overloaded, bool(isOverloaded(pingerrors.WithStack(err))))
	}
	invoke := func(context.Context, string, any, any, *grpc.ClientConn, ...grpc.CallOption) error { return nil }
	name := "github.com/tikv/pd/client/pkg/utils/grpcutil/triggerCircuitBreaker"
	require.NoError(t, failpoint.Enable(name, "return(true)"))
	defer failpoint.Disable(name)
	interceptor := UnaryCircuitBreakerInterceptor()
	require.NoError(t, interceptor(context.Background(), "/test", nil, nil, nil, invoke))
	disabled := circuitbreaker.NewCircuitBreaker("grpcutil-disabled", circuitbreaker.AlwaysClosedSettings)
	require.NoError(t, interceptor(circuitbreaker.WithCircuitBreaker(context.Background(), disabled), "/test", nil, nil, nil, invoke))
	settings := circuitbreaker.AlwaysClosedSettings
	settings.ErrorRateThresholdPct = 50
	enabled := circuitbreaker.NewCircuitBreaker("grpcutil-enabled", settings)
	err := interceptor(circuitbreaker.WithCircuitBreaker(context.Background(), enabled), "/test", nil, nil, nil, invoke)
	require.Equal(t, codes.ResourceExhausted, status.Code(err), "failpoint must be enabled in the isolated module")
}

func TestTransportContractCacheAndDial(t *testing.T) {
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	require.NoError(t, err)
	server := grpc.NewServer()
	healthpb.RegisterHealthServer(server, health.NewServer())
	done := make(chan struct{})
	go func() { defer close(done); _ = server.Serve(listener) }()
	defer func() { server.Stop(); listener.Close(); <-done }()
	addr := "http://" + listener.Addr().String()
	var connections sync.Map
	var wg sync.WaitGroup
	for range 8 {
		wg.Add(1)
		go func() {
			defer wg.Done()
			_, e := GetOrCreateGRPCConn(context.Background(), &connections, addr, nil, grpc.WithBlock())
			require.NoError(t, e)
		}()
	}
	wg.Wait()
	count := 0
	connections.Range(func(_, _ any) bool { count++; return true })
	require.Equal(t, 1, count)
	cached, err := GetOrCreateGRPCConn(context.Background(), &connections, addr, nil)
	require.NoError(t, err)
	defer cached.Close()
	_, err = healthpb.NewHealthClient(cached).Check(context.Background(), &healthpb.HealthCheckRequest{})
	require.NoError(t, err)
	require.NoError(t, cached.Close())
	again, err := GetOrCreateGRPCConn(context.Background(), &connections, addr, nil)
	require.NoError(t, err)
	require.Same(t, cached, again, "source cache returns even a closed connection")
	require.Equal(t, connectivity.Shutdown, again.GetState())

	_, err = GetClientConn(context.Background(), "http://%invalid", nil)
	require.Error(t, err)
	require.Contains(t, err.Error(), "PD:url:ErrURLParse")
	point := "github.com/tikv/pd/client/pkg/utils/grpcutil/unreachableNetwork2"
	require.NoError(t, failpoint.Enable(point, `return("http://%invalid")`))
	defer failpoint.Disable(point)
	_, err = GetOrCreateGRPCConn(context.Background(), &connections, "http://%invalid", nil)
	require.EqualError(t, err, "unreachable network")

	// grpcutil appends its connect parameters after caller options.
	var attempts atomic.Int32
	attempted := make(chan struct{}, 8)
	conn, err := GetClientConn(context.Background(), addr, nil, grpc.WithConnectParams(grpc.ConnectParams{Backoff: backoff.Config{BaseDelay: time.Millisecond, Multiplier: 1, MaxDelay: time.Millisecond}}), grpc.WithContextDialer(func(context.Context, string) (net.Conn, error) {
		attempts.Add(1)
		attempted <- struct{}{}
		return nil, errors.New("offline")
	}))
	require.NoError(t, err)
	defer conn.Close()
	select {
	case <-attempted:
	case <-time.After(3 * time.Second):
		t.Fatal("background dial did not start")
	}
	select {
	case <-attempted:
		t.Fatal("caller connect backoff incorrectly won over grpcutil's one-second first backoff")
	case <-time.After(200 * time.Millisecond):
	}
	require.Equal(t, int32(1), attempts.Load())
}

// The outer caller interceptor runs once; retry encloses the breaker, so
// each physical RPC contributes its own success/overload observation.
func TestTransportContractWireInterception(t *testing.T) {
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	require.NoError(t, err)
	var physical atomic.Int32
	server := grpc.NewServer(grpc.UnaryInterceptor(func(ctx context.Context, req any, _ *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (any, error) {
		if physical.Add(1) <= 2 {
			return nil, status.Error(codes.Unavailable, "retry me")
		}
		require.Equal(t, "leader:2379", GetForwardedHost(ctx, metadata.FromIncomingContext))
		return handler(ctx, req)
	}))
	healthpb.RegisterHealthServer(server, health.NewServer())
	done := make(chan struct{})
	go func() { defer close(done); _ = server.Serve(listener) }()
	defer func() { server.Stop(); listener.Close(); <-done }()
	var logical atomic.Int32
	conn, err := GetClientConn(context.Background(), "http://"+listener.Addr().String(), nil, grpc.WithBlock(), grpc.WithChainUnaryInterceptor(func(ctx context.Context, method string, req, reply any, cc *grpc.ClientConn, invoke grpc.UnaryInvoker, opts ...grpc.CallOption) error {
		logical.Add(1)
		return invoke(ctx, method, req, reply, cc, opts...)
	}))
	require.NoError(t, err)
	defer conn.Close()
	settings := circuitbreaker.AlwaysClosedSettings
	settings.ErrorRateThresholdPct = 50
	breaker := circuitbreaker.NewCircuitBreaker("grpcutil-wire", settings)
	overload := pdmetrics.CircuitBreakerCounters.WithLabelValues("grpcutil_wire", "overload")
	success := pdmetrics.CircuitBreakerCounters.WithLabelValues("grpcutil_wire", "success")
	beforeOverload, beforeSuccess := counterValue(t, overload), counterValue(t, success)
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	ctx = BuildForwardContext(ctx, "leader:2379")
	ctx = retry.WithBackoffer(ctx, retry.InitialBackoffer(time.Millisecond, time.Millisecond, 4*time.Millisecond))
	ctx = circuitbreaker.WithCircuitBreaker(ctx, breaker)
	_, err = healthpb.NewHealthClient(conn).Check(ctx, &healthpb.HealthCheckRequest{})
	require.NoError(t, err)
	require.Equal(t, int32(1), logical.Load())
	require.Equal(t, int32(3), physical.Load())
	require.Equal(t, float64(2), counterValue(t, overload)-beforeOverload)
	require.Equal(t, float64(1), counterValue(t, success)-beforeSuccess)
}

func counterValue(t *testing.T, counter prometheus.Counter) float64 {
	t.Helper()
	var metric dto.Metric
	require.NoError(t, counter.Write(&metric))
	return metric.GetCounter().GetValue()
}
