package grpcutil

import (
	"context"
	"net"
	"testing"
	"time"

	"github.com/stretchr/testify/require"
	"google.golang.org/grpc"
	"google.golang.org/grpc/connectivity"
)

func TestTransportContractPrefaceValidation(t *testing.T) {
	for _, test := range []struct {
		name       string
		frame      []byte
		fragmented bool
		ready      bool
	}{
		{"fragmented-settings-zero-capacity", []byte{0, 0, 6, 4, 0, 0, 0, 0, 0, 0, 3, 0, 0, 0, 0}, true, true},
		{"fragmented-settings-ack", []byte{0, 0, 0, 4, 1, 0, 0, 0, 0}, true, true},
		{"unknown-after-settings", []byte{0, 0, 0, 4, 0, 0, 0, 0, 0, 0, 0, 0, 255, 0, 0, 0, 0, 0}, false, true},
		{"ack-with-payload", []byte{0, 0, 6, 4, 1, 0, 0, 0, 0, 0, 3, 0, 0, 0, 1}, false, false},
		{"settings-on-stream-one", []byte{0, 0, 0, 4, 0, 0, 0, 0, 1}, false, false},
		{"invalid-settings-length", []byte{0, 0, 1, 4, 0, 0, 0, 0, 0, 0}, false, false},
	} {
		t.Run(test.name, func(t *testing.T) {
			listener, err := net.Listen("tcp", "127.0.0.1:0")
			require.NoError(t, err)
			defer listener.Close()
			ctx, cancel := context.WithTimeout(context.Background(), 500*time.Millisecond)
			defer cancel()
			result := make(chan *grpc.ClientConn, 1)
			go func() {
				conn, _ := GetClientConn(ctx, "http://"+listener.Addr().String(), nil, grpc.WithBlock())
				result <- conn
			}()
			socket := acceptOwnerPreface(t, listener)
			if test.fragmented {
				_, err = socket.Write(test.frame[:len(test.frame)-1])
				require.NoError(t, err)
				select {
				case conn := <-result:
					if conn != nil {
						conn.Close()
					}
					t.Fatal("setup completed before the entire SETTINGS frame arrived")
				case <-time.After(75 * time.Millisecond):
				}
				_, err = socket.Write(test.frame[len(test.frame)-1:])
			} else {
				_, err = socket.Write(test.frame)
			}
			require.NoError(t, err)
			conn := <-result
			if test.ready {
				require.NotNil(t, conn)
				defer conn.Close()
				require.Equal(t, connectivity.Ready, conn.GetState())
			} else {
				if conn != nil {
					defer conn.Close()
				}
				require.Nil(t, conn)
			}
		})
	}
}
