package util

import (
	"testing"
	"time"
)

// Run beside the unchanged master pkg/util/tokenlimiter.go. This is a
// focused ownership oracle, not the complete pkg/util test suite.
func TestCommandTokenOwnership(t *testing.T) {
	limiter := NewTokenLimiter(2)
	first, second := limiter.Get(), limiter.Get()
	acquired := make(chan *Token, 1)
	starting := make(chan struct{})
	go func() {
		close(starting)
		acquired <- limiter.Get()
	}()
	<-starting
	var early *Token
	select {
	case early = <-acquired:
	case <-time.After(20 * time.Millisecond):
	}
	// Another server owns a separate token channel.
	independent := NewTokenLimiter(1)
	independent.Put(independent.Get())
	limiter.Put(first)
	var third *Token
	if early != nil {
		third = early
	} else {
		select {
		case third = <-acquired:
		case <-time.After(time.Second):
			t.Fatal("returned token did not admit waiting command")
		}
	}
	limiter.Put(second)
	limiter.Put(third)
	if early != nil || len(limiter.ch) != 2 {
		t.Fatal("command bypassed admission or failed to return its token")
	}
	zero := NewTokenLimiter(0)
	if len(zero.ch) != 0 || cap(zero.ch) != 0 {
		t.Fatal("explicit zero is an empty channel, not unlimited admission")
	}
}
